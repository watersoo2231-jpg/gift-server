# 카페24 상세페이지 선물하기 버튼 삽입 가이드

## 서버 URL
```
https://gift-server-production-3f51.up.railway.app
```

## STEP 1 — 상품 상세 페이지에 선물하기 버튼 삽입

### 카페24 관리자 → 디자인 편집기 접속
1. 카페24 관리자 → **디자인(PC/모바일)** → **PC 디자인 편집** 클릭
2. 왼쪽 파일 목록 → `product` 폴더 → `product.html` 클릭
3. `Ctrl+F` 로 `바로구매` 또는 `btnBuy` 검색
4. 해당 버튼 코드 **바로 아래**에 아래 코드 붙여넣기:

```html
<!-- ===== 선물하기 버튼 시작 ===== -->
<button type="button" id="gift-btn" onclick="openGiftPopup()"
  style="display:block; width:100%; background:#ff5722; color:#fff; border:none; padding:15px; font-size:16px; font-weight:700; border-radius:8px; cursor:pointer; margin-top:10px;">
  🎁 선물하기
</button>

<!-- 선물하기 팝업 -->
<div id="gift-popup" style="display:none; position:fixed; top:0; left:0; width:100%; height:100%; background:rgba(0,0,0,0.6); z-index:99999; align-items:center; justify-content:center;">
  <div style="background:#fff; padding:32px 28px; border-radius:16px; width:90%; max-width:400px; box-shadow:0 8px 32px rgba(0,0,0,0.2);">
    <h3 style="text-align:center; margin-bottom:8px; font-size:18px;">🎁 선물하기</h3>
    <p style="text-align:center; color:#888; font-size:13px; margin-bottom:20px;">받는 분의 휴대폰 번호를 입력해 주세요</p>
    <input type="tel" id="gift-recipient-phone" placeholder="010-0000-0000"
      style="width:100%; padding:13px 14px; border:1px solid #ddd; border-radius:8px; font-size:15px; margin-bottom:12px; outline:none;">
    <input type="text" id="gift-message" placeholder="선물 메시지 (선택)"
      style="width:100%; padding:13px 14px; border:1px solid #ddd; border-radius:8px; font-size:15px; margin-bottom:16px; outline:none;">
    <button onclick="sendGift()"
      style="width:100%; background:#222; color:#fff; border:none; padding:14px; font-size:16px; font-weight:700; border-radius:8px; cursor:pointer; margin-bottom:10px;">
      선물 보내기
    </button>
    <button onclick="closeGiftPopup()"
      style="width:100%; background:#f5f5f5; color:#666; border:none; padding:12px; font-size:14px; border-radius:8px; cursor:pointer;">
      취소
    </button>
  </div>
</div>

<script>
var GIFT_SERVER = "https://gift-server-production-3f51.up.railway.app";

function openGiftPopup() {
  document.getElementById("gift-popup").style.display = "flex";
}
function closeGiftPopup() {
  document.getElementById("gift-popup").style.display = "none";
}
async function sendGift() {
  var phone = document.getElementById("gift-recipient-phone").value.trim();
  var message = document.getElementById("gift-message").value.trim();
  if (!phone) { alert("받는 분의 휴대폰 번호를 입력해 주세요."); return; }

  // 상품 정보 (카페24 변수로 자동 수집)
  var productId = "{product_no}" || "P001";
  var productName = "{product_name}" || "선물";
  var amount = parseInt("{product_price}") || 0;
  var senderName = "{member_name}" || "고객";

  var btn = document.querySelector("#gift-popup button");
  btn.disabled = true;
  btn.textContent = "처리 중...";

  try {
    var resp = await fetch(GIFT_SERVER + "/api/gift", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        recipient_phone: phone,
        recipient_email: phone + "@gift.local",
        product_id: productId,
        product_name: productName,
        amount: amount,
        sender_name: senderName,
        gift_message: message
      })
    });
    var result = await resp.json();
    if (result.success) {
      closeGiftPopup();
      alert("선물 발송 완료! 받는 분께 알림톡이 발송됩니다.");
    } else {
      alert("선물 보내기 실패: " + (result.error || "알 수 없는 오류"));
      btn.disabled = false;
      btn.textContent = "선물 보내기";
    }
  } catch(e) {
    alert("네트워크 오류가 발생했습니다. 다시 시도해 주세요.");
    btn.disabled = false;
    btn.textContent = "선물 보내기";
  }
}
</script>
<!-- ===== 선물하기 버튼 끝 ===== -->
```

5. **저장** 클릭

## STEP 3 — 주문완료 페이지 웹훅 연동 (선택)

`order` 폴더 → `order_result.html` 파일 맨 아래 `</body>` 위에 추가:

```html
<script>
// 선물 주문 완료 시 서버에 알림 (카페24 웹훅 대체)
(function() {
  var orderId = "{order_id}";
  var isGift = "{order_type}" === "gift" || document.referrer.includes("gift");
  if (!orderId || !isGift) return;

  fetch("https://gift-server-production-3f51.up.railway.app/webhook/cafe24/order", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      order_id: orderId,
      resource: {
        order_id: orderId,
        is_gift: "1",
        gift_receiver_name: "{gift_receiver_name}",
        gift_receiver_phone: "{gift_receiver_phone}",
        gift_message: "{gift_message}",
        buyer_name: "{member_name}"
      }
    })
  }).catch(function(){});
})();
</script>
```

## API 엔드포인트 정리

| 엔드포인트 | 메서드 | 설명 |
|-----------|--------|------|
| `/` | GET | 서버 상태 확인 |
| `/api/status` | GET | API 상태 확인 |
| `/api/gift` | POST | 선물 생성 |
| `/api/gift/{gift_id}` | GET | 선물 조회 |
| `/api/payment/create` | POST | 결제 생성 |
| `/api/payment/verify/{order_id}` | GET | 결제 검증 |
| `/api/payment/cancel/{order_id}` | POST | 결제 취소 |
| `/webhook/cafe24/order` | POST | 카페24 주문 웹훅 |
| `/gift/address` | GET | 배송지 입력 페이지 |
| `/gift/address` | POST | 배송지 저장 |
| `/admin/gifts` | GET | 선물 주문 현황 |
| `/oauth/install` | GET | 카페24 OAuth 인증 시작 |
